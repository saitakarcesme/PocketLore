cmake_minimum_required(VERSION 3.22)
project(pocketlore_attachments LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag BUILD_TRAINING_TOOLS BUILD_TESTS OPENMP_BUILD ENABLE_LTO ENABLE_NATIVE ENABLE_PRECOMPILED_HEADERS ENABLE_CCACHE WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER GGML_NATIVE GGML_OPENMP GGML_CUDA GGML_VULKAN GGML_BLAS GGML_METAL GGML_LLAMAFILE GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_BMI2 GGML_SSE42)
 set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(flag GRAPHICS_DISABLED DISABLED_LEGACY_ENGINE DISABLE_CURL DISABLE_ARCHIVE DISABLE_TIFF)
 set(${flag} ON CACHE BOOL "" FORCE)
endforeach()
set(CpuFeaturesNdkCompat_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CpuFeaturesNdkCompat")
add_library(leptonica STATIC IMPORTED GLOBAL)
set_target_properties(leptonica PROPERTIES IMPORTED_LOCATION "${ATTACHMENT_SOURCE}/install-${ANDROID_ABI}/lib/libleptonica.a")
add_subdirectory("${ATTACHMENT_SOURCE}/tesseract" tesseract)
add_subdirectory("${ATTACHMENT_SOURCE}/whisper" whisper)
add_library(pocketlore_attachments SHARED attachments.cpp)
target_link_libraries(pocketlore_attachments PRIVATE libtesseract whisper log)
target_include_directories(pocketlore_attachments PRIVATE "${ATTACHMENT_SOURCE}/tesseract/include" "${CMAKE_CURRENT_BINARY_DIR}/tesseract/include")
target_link_options(pocketlore_attachments PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,--exclude-libs,ALL")
